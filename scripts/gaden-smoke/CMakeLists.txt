cmake_minimum_required(VERSION 3.11)
project(plumenav_gaden_smoke LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(fmt REQUIRED)
if(NOT TARGET fmt)
  add_library(fmt ALIAS fmt::fmt)
endif()
set(CMAKE_DISABLE_FIND_PACKAGE_OpenCL TRUE)
set(BSC_ENABLE_CUDA OFF CACHE BOOL "" FORCE)
set(BSC_ENABLE_NATIVE_COMPILATION OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../research/external/gaden_core" gaden)
if(APPLE)
  target_include_directories(gaden PRIVATE /opt/homebrew/include)
  # libbsc ships a text file named 'version', which shadows libc++'s <version>
  # under ordinary -I search. Search that dependency after standard headers.
  foreach(prop INCLUDE_DIRECTORIES INTERFACE_INCLUDE_DIRECTORIES)
    get_target_property(dirs gaden ${prop})
    list(FILTER dirs EXCLUDE REGEX "/third_party/libbsc$")
    set_target_properties(gaden PROPERTIES ${prop} "${dirs}")
  endforeach()
  get_filename_component(bsc_root "${CMAKE_CURRENT_SOURCE_DIR}/../../research/external/gaden_core/third_party/libbsc" ABSOLUTE)
  target_compile_options(gaden PUBLIC -idirafter "${bsc_root}")
endif()
add_executable(plumenav_gaden_smoke smoke.cpp)
target_link_libraries(plumenav_gaden_smoke PRIVATE gaden)
